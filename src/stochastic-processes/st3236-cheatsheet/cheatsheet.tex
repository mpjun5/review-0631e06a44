\chapterhead*{Classification of States}

\begin{definition}[Accessibility]
State $j$ is \key{accessible} from state $i$, denoted by $i\to j$, if
\[ p_{ij}^{(n)}>0 \quad \text{for some } n\ge 0. \]
\end{definition}

\begin{remark}
\begin{itemize}\setlength\itemsep{0pt}\setlength\topsep{0pt}
\item Accessibility is not a one-step concept: $p_{ij}$ may be zero while $p_{ij}^{(n)}>0$ for some $n>1$.
\item It is not symmetric: $i\to j$ does not imply $j\to i$.
\item Since $p_{ii}^{(0)}=1$, every state is accessible from itself.
\end{itemize}
\end{remark}

\begin{definition}[Communication]
States $i$ and $j$ \key{communicate}, denoted by $i\longleftrightarrow j$, if
\[ i\to j \qquad\text{and}\qquad j\to i. \]
\end{definition}

\begin{note}
Communication is an \hl{equivalence} relation.
\end{note}

\begin{definition}[Communication Class]
A \key{communication class} is a maximal set $\mathcal{C}\subseteq S$ whose states communicate with one another.
\end{definition}

\begin{note}
Communication classes form a partition of the state space $S$.
\end{note}

\begin{definition}[Irreducible chains]
A Markov chain is \key{irreducible} if it has exactly one communication class. Otherwise, it is \key{reducible}.
\end{definition}

\begin{definition}[Closed Class]
A communication class $\mathcal{C}$ is \key{closed} if the chain cannot leave it:
\[ p_{ij}=0 \qquad \text{for every } i\in\mathcal{C} \text{ and } j\notin\mathcal{C}. \]
\end{definition}

\begin{remark}
\begin{itemize}\setlength\itemsep{0pt}\setlength\topsep{0pt}
\item Once $X_n\in\mathcal{C}$ for any $n$, all future states $X_{n+m}$ remain in $\mathcal{C}$ with probability 1, any $m\ge 0$.
\item An \key{absorbing state} $i$ satisfies $p_{ii}=1$. Hence $\{i\}$ is a closed class.
\item A closed class may contain several states and the chain may continue moving inside it.
\end{itemize}
\end{remark}

\begin{theorem}
Every finite-state Markov chain has at least one closed communicating class.
\end{theorem}

\begin{definition}[Hitting and First-Return Times]
For a state $j$, define
\[ T_j=\inf\{n\ge 0: X_n=j\}. \]
For a process \key{starting from state $i$}, define the \key{first \emph{return} time}
\[ T_i^{+}=\inf\{n\ge 1: X_n=i\}. \]
\end{definition}

\begin{remark}
\begin{itemize}\setlength\itemsep{0pt}\setlength\topsep{0pt}
\item $T_j$: the first time to arrive at state $j$.
\item $T_i^{+}$: we know that we start from $i$, and we wonder the first return time to $i$.
\end{itemize}
\end{remark}

\begin{definition}[First-Return Probability]
Starting from $i$, the probability that the first return occurs exactly at time $n$ is denoted as
\[ f_{ii}^{(n)}=P_i(T_i^{+}=n), \qquad n\ge 1. \]
The probability of ever returning is
\[ f_{ii}=P_i(T_i^{+}<\infty)=\sum_{n=1}^{\infty} f_{ii}^{(n)}. \]
\end{definition}

\begin{takeaway}
Consider state $i$. If the process returns to $i$ almost surely (with prob.\ 1), i.e.\ $f_{ii}=1$, then it will return to $i$ infinitely many times, and the total returning prob.\ $\sum\nolimits_{n=1}^{\infty}p_{ii}^{(n)}$ is infinite. We call $i$ \key{recurrent}. This is a class property.
\end{takeaway}

\begin{definition}[Recurrence and Transience]
A state $i$ is \key{recurrent} if
\[ {\color{csBlue} P_i(T_i^{+}<\infty)=1.} \]
It is \hl{transient} if
\[ {\color{csRed} P_i(T_i^{+}<\infty)<1.} \]
\end{definition}

Define the number of visits to $i$ after time zero:
\[ N_i=\sum_{n=1}^{\infty}\mathbf{1}\{X_n=i\}. \]

\begin{theorem}[Number of Revisits]
For every $m\ge 1$,
\[ {\color{csBlue} P_i(N_i\ge m)=(f_{ii})^m.} \]
Consequently,
\[ \begin{cases} N_i=\infty \ \text{a.s.,} & f_{ii}=1,\\ N_i<\infty \ \text{a.s.,} & f_{ii}<1. \end{cases} \]
\end{theorem}

\[ E_i[N_i]=\begin{cases} \dfrac{f_{ii}}{1-f_{ii}}, & f_{ii}<1,\\[4pt] \infty, & f_{ii}=1. \end{cases} \]
Hence,
\[ \begin{aligned}
{\color{csBlue} i \text{ recurrent}} &\iff N_i=\infty \text{ a.s.} \iff E_i[N_i]=\infty,\\
{\color{csRed} i \text{ transient}} &\iff N_i<\infty \text{ a.s.} \iff E_i[N_i]=\frac{f_{ii}}{1-f_{ii}}.
\end{aligned} \]

Define the $n$-step return probability
\[ p_{ii}^{(n)}=P_i(X_n=i). \]

\begin{theorem}[Return-Probability Criterion]
\[ \begin{aligned}
i \text{ is recurrent} &\iff \sum_{n=1}^{\infty}p_{ii}^{(n)}=\infty,\\
i \text{ is transient} &\iff \sum_{n=1}^{\infty}p_{ii}^{(n)}<\infty.
\end{aligned} \]
\end{theorem}

Starting from state $i$,
\[ \begin{aligned}
i \text{ recurrent} &\iff P_i(T_i^{+}<\infty)=f_{ii}=1\\
&\iff P_i(N_i=\infty)=1\\
&\iff E_i[N_i]=\infty\\
&\iff \sum_{n=1}^{\infty}p_{ii}^{(n)}=\infty.
\end{aligned} \]

\begin{remark}
\begin{itemize}\setlength\itemsep{0pt}\setlength\topsep{0pt}
\item $f_{ii}^{(n)}=P_i(T_i^{+}=n)$ counts paths whose \emph{first} return occurs at time $n$.
\item $p_{ii}^{(n)}=P_i(X_n=i)$ counts all paths that are at $i$ at time $n$, including paths that returned earlier.
\end{itemize}
\end{remark}

\begin{note}
So, $f_{ii}^{(n)}\le p_{ii}^{(n)}$.
\end{note}

\begin{theorem}
If $i\longleftrightarrow j$, then $i$ and $j$ are either both recurrent or both transient.
\end{theorem}

\begin{theorem}
For a \hl{finite-state} Markov chain:
\begin{itemize}\setlength\itemsep{0pt}\setlength\topsep{0pt}
\item at least \key{one closed communicating class} exists;
\item recurrent class $\iff$ closed class;
\item all states in a nonclosed class are transient.
\end{itemize}
\end{theorem}

\begin{definition}[Period]
For a state $i$, define
\[ {\color{csBlue} d(i)=\gcd\{n\ge 1: p_{ii}^{(n)}>0\}.} \]
If the set is empty, use the convention $d(i)=0$.\\
A state is \key{aperiodic} if $d(i)=1$.
\end{definition}

\begin{theorem}
If $i\longleftrightarrow j$, then
\[ {\color{csBlue} d(i)=d(j).} \]
\end{theorem}

\begin{definition}[Aperiodic Chain]
A Markov chain is \key{aperiodic} if every state is aperiodic, i.e.\ $d(i)=1$ for all $i\in S$.
\end{definition}

\begin{theorem}
Let state $i$ have period $d=d(i)\ge 1$.
\begin{itemize}\setlength\itemsep{0pt}\setlength\topsep{0pt}
\item For every $n\ge 1$,
\[ d\nmid n \quad\implies\quad p_{ii}^{(n)}=0. \]
\item There exists an integer $N_i\ge 1$ such that
\[ {\color{csBlue} p_{ii}^{(kd)}>0 \qquad \text{for every integer } k\ge N_i.} \]
\end{itemize}
\end{theorem}

\begin{note}
Some small multiples of $d$ may not be possible return times.
\end{note}

\begin{definition}[Regularity]
A \hl{finite-state} Markov chain with transition matrix $\mathbf{P}$ is \key{regular} if there exists an integer $r\ge 1$ such that \key{every entry of $\mathbf{P}^r$ is strictly positive}; that is,
\[ p_{ij}^{(r)}>0 \qquad \text{for all } i,j\in S. \]
\end{definition}

\begin{theorem}
For a finite-state Markov chain,
\[ {\color{csBlue} \text{regular} \quad\iff\quad \text{irreducible and aperiodic.}} \]
\end{theorem}

\begin{theorem}
Let $\{X_n\}$ be a \key{regular} Markov chain on a finite state space $S$, with transition matrix $\mathbf{P}$. There exists a distribution $\bpi=(\pi(j))_{j\in S}$ such that, for every initial distribution $\bpi_0$,
\[ \lim_{n\to\infty}\bpi_n=\lim_{n\to\infty}\bpi_0\mathbf{P}^n=\bpi. \]
\begin{itemize}\setlength\itemsep{0pt}\setlength\topsep{0pt}
\item The convergence is entrywise convergence, i.e.\ $\pi_n(j)\to\pi(j)$ for all $j\in S$.
\item This limit is independent of $\bpi_0$, i.e.\ $p_{ij}^{(n)}\to\pi(j)$ for all $i,j\in S$.
\item $\bpi$ is unique, and it satisfies
\[ \bpi\mathbf{P}=\bpi, \qquad \sum_{j\in S}\pi(j)=1, \qquad \pi(j)>0. \]
\end{itemize}
\end{theorem}

To calculate the limiting distribution, solve
\[ \bpi\mathbf{P}=\bpi, \qquad {\color{csBlue}\sum_{j\in S}\pi(j)=1,} \qquad \pi(j)\ge 0. \]
\begin{itemize}\setlength\itemsep{0pt}\setlength\topsep{0pt}
\item Before calculation, \hl{verify the convergence first!}
\item This is a \key{linear system} in the unknowns $(\pi(j))_{j\in S}$.
\end{itemize}

\begin{remark}
The $j$th entry $\pi(j)$ of $\bpi$ can be interpreted as the \key{long-run fraction of time spent in state $j$}:
\[ \left(\frac{1}{n}\sum_{k=1}^{n}\mathbf{1}_{\{X_k=j\}}\right)_{j\in S}\xrightarrow{\text{a.s.}}\bpi. \]
This convergence holds with probability 1.
\end{remark}

\begin{definition}[Positive and Null Recurrence]
For a recurrent state $j$, let $m_j=\E_j[T_j^+]$, where $T_j^+$ is the first return time to state $j$.
\begin{itemize}\setlength\itemsep{0pt}\setlength\topsep{0pt}
\item \key{Positive} recurrent: $m_j<\infty$.
\item \key{Null} recurrent: $m_j=\infty$.
\end{itemize}
\end{definition}

\begin{remark}
\begin{itemize}\setlength\itemsep{0pt}\setlength\topsep{0pt}
\item Recurrence guarantees a return, but the mean may be infinite. Positive recurrence further guarantees a finite return time.
\item Positive and null recurrence are class properties.
\item Every finite irreducible chain is positive recurrent.
\end{itemize}
\end{remark}

\begin{definition}[Ergodicity]
A Markov chain is \key{ergodic} if it is irreducible, \hl{positive recurrent}, and aperiodic.
\end{definition}

\begin{theorem}[Limiting Distribution]
An ergodic chain on a finite or countable state space $S$ has a unique limiting distribution $\bpi=(\pi(j))_{j\in S}$. For every initial distribution $\bpi_0$,
\[ \lim_{n\to\infty}\pi_n(j)=\pi(j)=\frac{1}{m_j}>0, \qquad j\in S. \]
\begin{itemize}\setlength\itemsep{0pt}\setlength\topsep{0pt}
\item The limit is independent of $\bpi_0$.
\item It satisfies
\[ \bpi\mathbf{P}=\bpi, \qquad \sum_{j\in S}\pi(j)=1. \]
\end{itemize}
\end{theorem}
\begin{itemize}\setlength\itemsep{0pt}\setlength\topsep{0pt}
\item For finite chains, ergodicity is equivalent to regularity.
\item Irreducible $+$ aperiodic $+$ finite states $\Rightarrow$ limiting dist.
\item Irreducible $+$ aperiodic $+$ positive recurrent $\Rightarrow$ limiting dist.
\end{itemize}

\begin{itemize}\setlength\itemsep{1pt}\setlength\topsep{2pt}
\item \textbf{Convergence.} Finite chains: regular, equivalently irreducible and aperiodic. Countable chains: irreducible, positive recurrent and aperiodic. Then $\bpi_n\to\bpi$, independently of $\bpi_0$.
\item \textbf{Calculation.} Verify convergence first; then solve $\bpi\mathbf{P}=\bpi$, $\sum\nolimits_j\pi(j)=1$, $\pi(j)\ge0$. For an ergodic chain, $\pi(j)=1/m_j$.
\item \textbf{Interpretation.} In an ergodic chain, fixed-time probabilities and one-path visit frequencies both approach $\pi(j)$.
\item \textbf{Periodicity and null recurrence.} Positive recurrent periodic chains retain phase limits and time averages, including one-path frequencies. For a null recurrent state $j$, $\pi_n(j)\to0$.
\item \textbf{Multiple classes.} Combine entering probabilities with within-class behaviour. The starting state can affect the limit.
\end{itemize}

\begin{cstable}{lCCC}
 & \textbf{Transient} & \textbf{Null recurrent} & \textbf{Positive recurrent} \\ \midrule
$\mathbb{P}_i(T_i^+<\infty)$ & $f_{ii}<1$ & $1$ & $1$ \\
$N_i$ & finite a.s. & $\infty$ a.s. & $\infty$ a.s. \\
$m_i=\E_i[T_i^+]$ & $\infty$ & $\infty$ & finite \\
$\E_i[N_i]$ & $f_{ii}/(1-f_{ii})$ & $\infty$ & $\infty$ \\

\end{cstable}

\chapterhead*{Stationary Distributions}

\begin{definition}[Stationary Distribution]
A probability vector $\bpi=(\pi(j))_{j\in S}$ is a \key{stationary distribution} for $\mathbf{P}$ if ${\color{csBlue}\bpi=\bpi\mathbf{P}}$.
\end{definition}

\begin{definition}[Global Balance Equations]
A probability vector $\bpi$ satisfies \textbf{global balance} if, for every $j$,
\[ {\color{csRed}\underbrace{\pi(j)}_{\text{Leaving prob.}}} = {\color{csBlue}\underbrace{\sum_i \pi(i)p_{ij}}_{\text{Arriving prob.}}}\,. \]
Equivalently, excluding self-transitions, ${\color{csRed}\pi(j)\sum\nolimits_{k\ne j}p_{jk}} = {\color{csBlue}\sum\nolimits_{i\ne j}\pi(i)p_{ij}}$.
\end{definition}

\begin{definition}[Detailed Balance Equations]
A probability vector $\bpi$ satisfies the \key{detailed balance equations} if, for every pair $i,j$,
\[ {\color{csBlue}\pi(i)p_{ij}=\pi(j)p_{ji}}. \]
\end{definition}

\begin{note}
Detailed balance $\Rightarrow$ global balance, but the converse is not true.
\end{note}

\begin{theorem}
For an \key{irreducible} chain, the existence of a stationary probability distribution implies \hl{positive recurrence}.
\end{theorem}
