\chapterhead*{Classification of States}

\begin{definition}[Accessibility]
State $j$ is \key{accessible} from state $i$, denoted by $i\to j$, if
\[ p_{ij}^{(n)}>0 \quad \text{for some } n\ge 0. \]
\end{definition}

\begin{remark}
\begin{itemize}\setlength\itemsep{0pt}\setlength\topsep{0pt}
\item Accessibility is not a one-step concept: $p_{ij}$ may be zero while $p_{ij}^{(n)}>0$ for some $n>1$.
\item It is not symmetric: $i\to j$ does not imply $j\to i$.
\item Since $p_{ii}^{(0)}=1$, every state is accessible from itself.
\end{itemize}
\end{remark}

\begin{definition}[Communication]
States $i$ and $j$ \key{communicate}, denoted by $i\longleftrightarrow j$, if
\[ i\to j \qquad\text{and}\qquad j\to i. \]
\end{definition}

\begin{note}
Communication is an \hl{equivalence} relation.
\end{note}

\begin{definition}[Communication Class]
A \key{communication class} is a maximal set $\mathcal{C}\subseteq S$ whose states communicate with one another.
\end{definition}

\begin{note}
Communication classes form a partition of the state space $S$.
\end{note}

\begin{definition}[Irreducible chains]
A Markov chain is \key{irreducible} if it has exactly one communication class. Otherwise, it is \key{reducible}.
\end{definition}

\begin{definition}[Closed Class]
A communication class $\mathcal{C}$ is \key{closed} if the chain cannot leave it:
\[ p_{ij}=0 \qquad \text{for every } i\in\mathcal{C} \text{ and } j\notin\mathcal{C}. \]
\end{definition}

\begin{remark}
\begin{itemize}\setlength\itemsep{0pt}\setlength\topsep{0pt}
\item Once $X_n\in\mathcal{C}$ for any $n$, all future states $X_{n+m}$ remain in $\mathcal{C}$ with probability 1, any $m\ge 0$.
\item An \key{absorbing state} $i$ satisfies $p_{ii}=1$. Hence $\{i\}$ is a closed class.
\item A closed class may contain several states and the chain may continue moving inside it.
\end{itemize}
\end{remark}

\begin{theorem}
Every finite-state Markov chain has at least one closed communicating class.
\end{theorem}
