\defQ{9}{}{%
Solve the system of equations
\[\begin{aligned}
2x^{3}+3\ln y-\e^{z}&=18,\\
x^{3}-\ln y+2\e^{z}&=9,\\
x^{3}+2\ln y+\e^{z}&=11,
\end{aligned}\]
where $x$, $y$ and $z$ are real and $y>0$. \mk{4}
\src{Example}}
\defS{9}{%
\begin{topalign}
\text{Let } A=x^{3},\ B&=\ln y,\ C=\e^{z}\\
2A+3B-C&=18 &&---(1)\\
A-B+2C&=9 &&---(2)\\
A+2B+C&=11 &&---(3)
\end{topalign}
\topdisplay{\text{Using GC, } A=8,\ B=1,\ C=1}
\begin{topalign}
x^{3}&=8 &\ln y&=1 &\e^{z}&=1\\
\therefore\ x&=2 &\therefore\ y&=\e &\therefore\ z&=0
\end{topalign}}
