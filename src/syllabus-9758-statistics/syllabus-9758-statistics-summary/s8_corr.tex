\section{Product Moment Correlation Coefficient $r$}
\begin{theorem}[before skip=2mm,after skip=3mm]{Product Moment Correlation Coefficient (PMCC) (MF27)}
\centering$r=\dfrac{\sum(x-\xb)(y-\yb)}{\sqrt{\sum(x-\xb)^{2}\sum(y-\yb)^{2}}}=\dfrac{\Sxy}{\sqrt{\Sxx\Syy}}$\par\medskip
$\Sxx=\displaystyle\sum x^{2}-\dfrac{\left(\sum x\right)^{2}}{n},\quad\Syy=\displaystyle\sum y^{2}-\dfrac{\left(\sum y\right)^{2}}{n},\quad\Sxy=\displaystyle\sum xy-\dfrac{\sum x\sum y}{n}$
\end{theorem}
\begin{center}\ba\small
\begin{tabular}{|M{3.2cm}|L{12.0cm}|}\hline
\textbf{Value of $r$} & \textbf{Interpretation}\\\Xhline{2pt}
$r=1$ or $-1$ & all points lie exactly on a straight line (positive / negative gradient)\\\hline
$r$ close to $1$ & strong positive linear correlation\\\hline
$r$ close to $-1$ & strong negative linear correlation\\\hline
$r$ close to $0$ & little or no \textbf{linear} correlation (a non-linear relation may still exist)\\\hline
\end{tabular}\end{center}
\begin{info}[title=Note]
$-1\leqslant r\leqslant1$. \ $|r|$ close to $1$ only says a \textbf{linear} model fits well; check the scatter diagram. A strong correlation does \textbf{not} mean that $x$ causes $y$.
\end{info}

\cdisp{r&=\dfrac{\sum xy-\dfrac{\sum x\sum y}{n}}{\sqrt{\left(\sum x^{2}-\dfrac{\left(\sum x\right)^{2}}{n}\right)\left(\sum y^{2}-\dfrac{\left(\sum y\right)^{2}}{n}\right)}}\\&=\dfrac{143.4-\dfrac{21(35.4)}{6}}{\sqrt{\left(91-\dfrac{21^{2}}{6}\right)\left(231-\dfrac{35.4^{2}}{6}\right)}}\\&=\dfrac{19.5}{\sqrt{17.5\times22.14}}\\\therefore\ r&=0.991\ \text{(3 s.f.)}}
\section{Interpolation, Extrapolation and Interpretation}
\begin{center}\ba\small
\begin{tabular}{|L{4.0cm}|L{11.2cm}|}\hline
\textbf{Item} & \textbf{Note}\\\Xhline{2pt}
Interpolation & estimate for $x$ \textbf{within} the data range: reliable if $|r|$ is close to $1$ and the scatter diagram is linear\\\hline
Extrapolation & estimate for $x$ \textbf{outside} the data range: unreliable, the linear relation may not continue\\\hline
Gradient $b$ & the change in $y$ per unit increase in $x$ (in context)\\\hline
Intercept $a$ & value of $y$ when $x=0$; meaningful only if $x=0$ is sensible in context\\\hline
How well does the model fit? & $|r|$ close to $1$ and points close to the line $\Rightarrow$ good fit; comment on the range of $x$\\\hline
\end{tabular}\end{center}
\begin{info}[title=Note]
Reliability of an estimate requires \textbf{two} conditions:
\begin{itemize}[leftmargin=1.4em,itemsep=0pt,topsep=2pt]
\item interpolation is used (state the data range);
\item $|r|$ is close to $1$, indicating a strong linear correlation between the two variables.
\end{itemize}
If any one of these fails, the estimate is unreliable.
\end{info}

\section{Sum of Squares of Residuals}
\begin{theorem}[before skip=2mm,after skip=3mm]{Residual}
For a line $y=f(x)$ fitted to data, the residual at the data point $(x_i,y_i)$ is
\[e_i=y_i-f(x_i)\qquad\text{(observed value $-$ fitted value)}.\]
\end{theorem}
\noindent\exlab The least squares regression line for the 6 data points $(1,3.1)$, $(2,4.3)$, $(3,5.2)$, $(4,6.8)$, $(5,7.1)$, $(6,8.9)$ is $y=2.00+1.11x$. Find the residuals and the sum of squares of residuals.
\begin{center}\begin{tikzpicture}\begin{axis}[sc,width=6.5cm,height=4.6cm,xmin=0,xmax=7,ymin=0,ymax=10]
\addplot[black,semithick,domain=0.3:6.7]{1.11*x+2};
\draw[blue,thick,dotted](axis cs:1,3.1)--(axis cs:1,3.11);
\draw[blue,thick,dotted](axis cs:2,4.3)--(axis cs:2,4.22);
\draw[blue,thick,dotted](axis cs:3,5.2)--(axis cs:3,5.33);
\draw[blue,thick,dotted](axis cs:4,6.8)--(axis cs:4,6.44);
\draw[blue,thick,dotted](axis cs:5,7.1)--(axis cs:5,7.55);
\draw[blue,thick,dotted](axis cs:6,8.9)--(axis cs:6,8.66);
\addplot[only marks,mark=*,mark size=1.2pt]coordinates{(1,3.1)(2,4.3)(3,5.2)(4,6.8)(5,7.1)(6,8.9)};
\node[font=\scriptsize,anchor=west] at (axis cs:4.3,9.3){$y=2.00+1.11x$};
\end{axis}\end{tikzpicture}\end{center}
\cdisp{e_1&=3.1-(2.00+1.11(1))=-0.01\\e_2&=4.3-(2.00+1.11(2))=0.08\\e_3&=5.2-(2.00+1.11(3))=-0.13\\e_4&=6.8-(2.00+1.11(4))=0.36\\e_5&=7.1-(2.00+1.11(5))=-0.45\\e_6&=8.9-(2.00+1.11(6))=0.24}
\cdisp{\text{Sum of squares of residuals}&=(3.1-3.11)^{2}+(4.3-4.22)^{2}+(5.2-5.33)^{2}\\&\quad+(6.8-6.44)^{2}+(7.1-7.55)^{2}+(8.9-8.66)^{2}\\&=(-0.01)^{2}+(0.08)^{2}+(-0.13)^{2}+(0.36)^{2}+(-0.45)^{2}+(0.24)^{2}\\\therefore\ \textstyle\sum e_i^{2}&=0.4131}
\begin{info}[title=Note]
Let $c$ be the sum of squares of residuals of the least squares regression line. For any line $y=a+bx$,
\[\sum_{i=1}^{n}\bigl(y_i-(a+bx_i)\bigr)^{2}\geqslant c,\]
with equality if and only if $y=a+bx$ is the least squares regression line.
\end{info}
\begin{info}[title=Note]
\textbf{The residuals are squared} because some points are above the line, some below and some on the line. The residuals may be positive, negative or zero; adding them causes them to cancel out. Squaring them ensures they are all non-negative.\par\smallskip
\textbf{The least squares regression line is the line that minimises the sum of squares of residuals.}
\end{info}

Extra point that leaves the line unchanged & the point $(\xb,\yb)$ (the mean point)\\\hline
Change of units or coding, $x'=px+q$ & $r$ unchanged if $p>0$ (sign of $r$ reverses if $p<0$); new line: substitute $x=\dfrac{x'-q}{p}$ into the fitted equation\\\hline
