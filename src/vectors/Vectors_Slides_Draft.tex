\begin{frame}{Vector product: perpendicular to both}
\twocol{
\begin{tikzpicture}[v3=1.9]
\coordinate (O) at (0,0,0); \coordinate (A) at (2,0,0); \coordinate (B) at (0.8,1.8,0);
\fill[vP,opacity=0.5] (O)--(A)--(2.8,1.8,0)--(B)--cycle;
\draw[va] (O)--(A) node[right]{$\mathbf{a}$};
\draw[vb] (O)--(B) node[above right]{$\mathbf{b}$};
\draw[vn] (O)--(0,0,2.2) node[above]{$\mathbf{a}\times\mathbf{b}$};
\draw[vn, dashed] (O)--(0,0,-1.4) node[below]{$\mathbf{b}\times\mathbf{a}$};
\end{tikzpicture}}{
\[
\mathbf{a}\times\mathbf{b} = \col{a_2b_3-a_3b_2}{-(a_1b_3-a_3b_1)}{a_1b_2-a_2b_1}
\]
\begin{align*}
(\mathbf{a}\times\mathbf{b})\cdot\mathbf{a} &= 0\\
\mathbf{b}\times\mathbf{a} &= -\mathbf{a}\times\mathbf{b}
\end{align*}}
\end{frame}

\begin{frame}{Area}
\twocol{
\begin{tikzpicture}[v3=1.8]
\coordinate (O) at (0,0,0); \coordinate (A) at (2.4,0,0.3); \coordinate (B) at (0.8,1.8,1); \coordinate (C) at (3.2,1.8,1.3);
\fill[vP,opacity=0.6] (O)--(A)--(C)--(B)--cycle;
\fill[vR,opacity=0.9] (O)--(A)--(B)--cycle;
\draw[va] (O)--(A) node[right]{$\mathbf{a}$}; \draw[vb] (O)--(B) node[left]{$\mathbf{b}$};
\draw[dsh] (A)--(C)--(B); \draw[thick] (A)--(B);
\end{tikzpicture}}{
\begin{align*}
\text{parallelogram} &= |\mathbf{a}\times\mathbf{b}|\\
\triangle OAB &= \tfrac12|\mathbf{a}\times\mathbf{b}|\\
\triangle ABC &= \tfrac12|\overrightarrow{AB}\times\overrightarrow{AC}|
\end{align*}}
\end{frame}
