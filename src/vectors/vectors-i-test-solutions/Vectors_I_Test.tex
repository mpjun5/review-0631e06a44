\defQ{tf}{}{%
For each of the following statements, state whether it is \textbf{true} or \textbf{false}.\par\smallskip
\noindent\fbox{\parbox{\dimexpr\linewidth-2\fboxsep-2\fboxrule\relax}{\centering\textbf{Each correct answer scores 1 mark.\\ A blank answer scores 0 marks.\\ Each wrong answer scores $-1$ mark.}\par}}\mk{15}
\begin{tftable}
\tfrow{$(\mathbf a+\mathbf b)\cdot(\mathbf a-\mathbf b)=|\mathbf a|^2-|\mathbf b|^2$.}{T}{$\mathbf a\cdot\mathbf a-\mathbf a\cdot\mathbf b+\mathbf b\cdot\mathbf a-\mathbf b\cdot\mathbf b=|\mathbf a|^2-|\mathbf b|^2$}
\tfrow{$|\mathbf a+\mathbf b|^2=|\mathbf a|^2+2\,\mathbf a\cdot\mathbf b+|\mathbf b|^2$.}{T}{$(\mathbf a+\mathbf b)\cdot(\mathbf a+\mathbf b)=\mathbf a\cdot\mathbf a+2\,\mathbf a\cdot\mathbf b+\mathbf b\cdot\mathbf b$}
\tfrow{$|\mathbf a+\mathbf b|^2=|\mathbf a|^2+2|\mathbf a||\mathbf b|+|\mathbf b|^2$.}{F}{$2\,\mathbf a\cdot\mathbf b=2|\mathbf a||\mathbf b|\cos\theta;\ \mathbf a=\mathbf i,\ \mathbf b=\mathbf j:\ |\mathbf i+\mathbf j|^2=2\neq4$}
\tfrow{$\mathbf a\times\mathbf a=\mathbf a^2$.}{F}{$\mathbf a\times\mathbf a=\mathbf 0\ \text{(zero vector)}$}
\tfrow{$\mathbf a\times\mathbf b=|\mathbf a||\mathbf b|\sin\theta$, where $\theta$ is the angle between $\mathbf a$ and $\mathbf b$.}{F}{$|\mathbf a\times\mathbf b|=|\mathbf a||\mathbf b|\sin\theta;\ \mathbf a\times\mathbf b\ \text{is a vector}$}
\tfrow{$\mathbf b\times\mathbf a=-\mathbf a\times\mathbf b$.}{T}{$\mathbf j\times\mathbf i=-\mathbf k=-(\mathbf i\times\mathbf j)$}
\tfrow{$(\mathbf a+\mathbf b)\times(\mathbf a-\mathbf b)=2(\mathbf b\times\mathbf a)$.}{T}{$\mathbf a\times\mathbf a-\mathbf a\times\mathbf b+\mathbf b\times\mathbf a-\mathbf b\times\mathbf b=\mathbf 0+\mathbf b\times\mathbf a+\mathbf b\times\mathbf a-\mathbf 0$}
\tfrow{$(\mathbf a+\mathbf b)\times(\mathbf a-\mathbf b)=|\mathbf a|^2-|\mathbf b|^2$.}{F}{$(\mathbf a+\mathbf b)\times(\mathbf a-\mathbf b)=2(\mathbf b\times\mathbf a)\ \text{(a vector)}$}
\tfrow{$\mathbf a\cdot(\mathbf a\times\mathbf b)=0$.}{T}{$\mathbf a\times\mathbf b\perp\mathbf a\ \therefore\ \mathbf a\cdot(\mathbf a\times\mathbf b)=0$}
\tfrow{$|\mathbf a\times\mathbf b|^2+(\mathbf a\cdot\mathbf b)^2=|\mathbf a|^2|\mathbf b|^2$.}{T}{$|\mathbf a|^2|\mathbf b|^2(\sin^2\theta+\cos^2\theta)=|\mathbf a|^2|\mathbf b|^2$}
\tfrow{$\mathbf a\cdot\mathbf b=\mathbf b\cdot\mathbf a$ and $\mathbf a\times\mathbf b=\mathbf b\times\mathbf a$.}{F}{$\mathbf b\times\mathbf a=-\mathbf a\times\mathbf b;\ \mathbf i\times\mathbf j=\mathbf k,\ \mathbf j\times\mathbf i=-\mathbf k$}
\tfrow{For non-zero $\mathbf a$ and $\mathbf b$, if $\mathbf a\cdot\mathbf b=0$, then $\mathbf a$ is perpendicular to $\mathbf b$.}{T}{$|\mathbf a||\mathbf b|\cos\theta=0\ \therefore\ \cos\theta=0,\ \theta=90^\circ$}
\tfrow{If $\mathbf a=3\mathbf b$, then $\mathbf a\times\mathbf b=\mathbf 0$.}{T}{$3\mathbf b\times\mathbf b=3(\mathbf b\times\mathbf b)=\mathbf 0$}
\tfrow{For non-zero $\mathbf a$, $|\mathbf a|\,\mathbf a$ is a unit vector.}{F}{$\bigl||\mathbf a|\,\mathbf a\bigr|=|\mathbf a|^2;\ \hat{\mathbf a}=\tfrac{1}{|\mathbf a|}\mathbf a$}
\tfrow{$\mathbf a\times\mathbf b$ and $\mathbf b\times\mathbf a$ point in the same direction.}{F}{$\mathbf b\times\mathbf a=-(\mathbf a\times\mathbf b)\ \text{(opposite direction)}$}
\end{tftable}}

\defQ{1}{DHS 2023 Promo Q5}{%
With reference to the origin $O$, the points $A$ and $B$ have coordinates $(1,2,0)$ and $(-5,1,0)$.
\begin{qparts}
\pt{(a)} Find the angle $AOB$.\mk{3}
\end{qparts}
The vectors $\vv{OA}$ and $\vv{OB}$ are projected along the direction of $\cvv{1}{2}{2}$ and $\cvv{2}{1}{2}$ to form the vectors $\vv{O'A'}$ and $\vv{O'B'}$ respectively.
\begin{qparts}
\pt{(b)} Find the vectors $\vv{O'A'}$ and $\vv{O'B'}$.\mk{3}
\pt{(c)} Find the exact area of the triangle $O'A'B'$.\mk{2}
\end{qparts}}

\defS{1}{%
\pt{(a)} $\vv{OA}=\cvv{1}{2}{0}$ and $\vv{OB}=\cvv{-5}{1}{0}$. Let $\theta=\angle AOB$.
\begin{align*}
\vv{OA}\cdot\vv{OB}&=|\vv{OA}||\vv{OB}|\cos\theta\\
\cos\theta&=\frac{\cvv{1}{2}{0}\cdot\cvv{-5}{1}{0}}{\sqrt{1^2+2^2}\sqrt{5^2+1^2}}\\
&=\frac{-5+2}{\sqrt5\sqrt{26}}=-\frac3{\sqrt5\sqrt{26}}\\
\theta&=105.3^\circ.
\end{align*}
\pt{(b)}
\begin{align*}
\vv{O'A'}&=\frac{\vv{OA}\cdot\cvv{1}{2}{2}}{\left|\cvv{1}{2}{2}\right|}\frac{\cvv{1}{2}{2}}{\left|\cvv{1}{2}{2}\right|}\\
&=\frac{\cvv{1}{2}{0}\cdot\cvv{1}{2}{2}}{(\sqrt{1^2+2^2+2^2})^2}\cvv{1}{2}{2}=\frac59\cvv{1}{2}{2}\\
\vv{O'B'}&=\frac{\vv{OB}\cdot\cvv{2}{1}{2}}{\left|\cvv{2}{1}{2}\right|}\frac{\cvv{2}{1}{2}}{\left|\cvv{2}{1}{2}\right|}\\
&=\frac{\cvv{-5}{1}{0}\cdot\cvv{2}{1}{2}}{(\sqrt{2^2+1^2+2^2})^2}\cvv{2}{1}{2}=-\cvv{2}{1}{2}.
\end{align*}
\pt{(c)}
\begin{align*}
\text{Area of triangle }O'A'B'&=\frac12|\vv{O'A'}\times\vv{O'B'}|\\
&=\frac5{18}\left|\cvv{1}{2}{2}\times\cvv{2}{1}{2}\right|\qquad(|\mathbf a\times(-\mathbf b)|=|\mathbf a\times\mathbf b|)\\
&=\frac5{18}\left|\cvv{2}{2}{-3}\right|\\
&=\frac5{18}\sqrt{17}\text{ units}^2.
\end{align*}}

\defQ{2}{NYJC 2024 Promo Q5}{%
The position vectors of points $A$, $B$ and $C$, relative to the origin $O$ are $\mathbf a$, $\mathbf b$ and $\mathbf c$ respectively, where
\[\mathbf a=3\mathbf i+2\mathbf j-\mathbf k,\quad \mathbf b=4\mathbf i-3\mathbf j+2\mathbf k,\quad \mathbf c=3\mathbf i-\mathbf j-\mathbf k.\]
\begin{qparts}
\pt{(a)} Find $\mathbf a\times\mathbf b$ and deduce the exact area of triangle $OAB$.\mk{3}
\pt{(b)} Hence, find the volume of the tetrahedron $OABC$.\par
[The volume of a tetrahedron is $\dfrac13\times\text{base area}\times\text{perpendicular height}$.]\mk{3}
\pt{(c)} Find the perpendicular distance from $B$ to line $OA$.\mk{2}
\end{qparts}}

\defS{2}{%
\pt{(a)}
\begin{align*}
\mathbf a\times\mathbf b&=\cvv{3}{2}{-1}\times\cvv{4}{-3}{2}=\cvv{1}{-10}{-17}\\
\text{Area of }\triangle AOB&=\frac12|\mathbf a\times\mathbf b|\\
&=\frac12\left|\cvv{1}{-10}{-17}\right|\\
&=\frac12\sqrt{1+100+289}=\frac12\sqrt{390}\text{ units}^2.
\end{align*}
\pt{(b)}
\begin{align*}
&\text{Perpendicular height of tetrahedron}\\
&=\frac1{\sqrt{1+100+289}}\left|\vv{OC}\cdot\cvv{1}{-10}{-17}\right|\\
&=\frac1{\sqrt{390}}\left|\cvv{3}{-1}{-1}\cdot\cvv{1}{-10}{-17}\right|\\
&=\frac{30}{\sqrt{390}}\\
&\text{Volume of tetrahedron }OABC\\
&=\frac13\left(\frac{\sqrt{390}}2\right)\left(\frac{30}{\sqrt{390}}\right)\\
&=5\text{ units}^3.
\end{align*}
\pt{(c)}
\begin{align*}
\text{Perpendicular distance from }B\text{ to line }OA
&=|\mathbf b\times\hat{\mathbf a}|\\
&=\frac{\sqrt{390}}{\sqrt{9+4+1}}\\
&=\sqrt{\frac{195}{7}}\quad\text{or }5.28.
\end{align*}}

\defQ{3}{MI 2024 Promo Q4}{%
Referred to the origin $O$, the points $A$, $B$ and $C$ have position vectors $\mathbf a$, $\mathbf b$ and $\mathbf c$. $C$ lies on $AB$ such that $AC:AB=1:3$.
\begin{qparts}
\pt{(i)} Find $\mathbf c$ in terms of $\mathbf a$ and $\mathbf b$.\mk{2}
\pt{(ii)} Show that the area of triangle $OAC$ is given by $k|\mathbf a\times\mathbf b|$, where $k$ is a constant to be determined.\mk{2}
\end{qparts}
It is now given that $\mathbf a$ and $\mathbf b$ are unit vectors.
\begin{qparts}
\pt{(iii)} Using the result in part (i), find the value of $(2\mathbf a-\mathbf b)\cdot\mathbf c$.\mk{3}
\end{qparts}}

\defS{3}{%
\pt{(i)}
\begin{align*}
AC:AB&=1:3\quad\Rightarrow\quad AC:CB=1:2\\
\mathbf c&=\frac{(1)\mathbf b+(2)\mathbf a}{1+2}=\frac{2\mathbf a+\mathbf b}{3}.
\end{align*}
\pt{(ii)}
\begin{align*}
\text{Area of triangle }OAC
&=\frac13\text{ Area of triangle }OAB\\
&=\frac13\times\frac12|\vv{OA}\times\vv{OB}|\\
&=\frac16|\mathbf a\times\mathbf b|,\qquad k=\frac16.
\end{align*}
\pt{(iii)}
\begin{align*}
(2\mathbf a-\mathbf b)\cdot\mathbf c
&=(2\mathbf a-\mathbf b)\cdot\left(\frac{2\mathbf a+\mathbf b}{3}\right)\\
&=\frac13(2\mathbf a-\mathbf b)\cdot(2\mathbf a+\mathbf b)\\
&=\frac13(4\mathbf a\cdot\mathbf a+2\mathbf a\cdot\mathbf b-2\mathbf b\cdot\mathbf a-\mathbf b\cdot\mathbf b)\\
&=\frac13(4|\mathbf a|^2-|\mathbf b|^2)\\
&=\frac13\left(4(1)^2-(1)^2\right)\quad\text{since $\mathbf a$, $\mathbf b$ are unit vectors}\\
&=1.
\end{align*}}

\defQ{4}{TMJC 2024 Promo Q6}{%
\begin{qparts}
\pt{(a)} The points $A$, $B$ and $C$ have position vectors $\cvv{1}{3}{-7}$, $\cvv{5}{1}{1}$ and $\cvv{-9}{8}{-27}$, with respect to the origin $O$.
\begin{qparts}[2.2em]
\pt{(i)} Show that the points $A$, $B$ and $C$ are collinear.\mk{2}
\pt{(ii)} Find the exact area of the triangle $OAB$.\mk{3}
\end{qparts}
\pt{(b)} The vectors $\mathbf u$, $\mathbf v$ and $\mathbf w$ are defined such that $|\mathbf u|=1$, $|\mathbf v|=2$ and $|\mathbf w|=3$.
\begin{qparts}[2.2em]
\pt{(i)} Suppose that $\mathbf u+\mathbf v=\mathbf w$. With the aid of a diagram, or otherwise, explain why $\mathbf u$ is a unit vector that is parallel to $\mathbf w$.\mk{2}
\pt{(ii)} Given instead that each of the three vectors is perpendicular to the sum of the other two vectors, find the exact value of $|\mathbf u+\mathbf v+\mathbf w|$.\mk{4}
\end{qparts}\end{qparts}}

\defS{4}{%
\pt{(a)(i)}
\begin{align*}
\vv{AB}&=\cvv{5}{1}{1}-\cvv{1}{3}{-7}=\cvv{4}{-2}{8}\\
\vv{AC}&=\cvv{-9}{8}{-27}-\cvv{1}{3}{-7}=\cvv{-10}{5}{-20}\\
&=-\frac52\cvv{4}{-2}{8}=-\frac52\vv{AB}.
\end{align*}
Since $\vv{AB}$ is parallel to $\vv{AC}$, with a common point $A$, the points $A$, $B$ and $C$ are collinear. (Shown)\par
\pt{(a)(ii)}
\begin{align*}
\text{Area of triangle }OAB&=\frac12|\vv{OA}\times\vv{OB}|\\
&=\frac12\left|\cvv{1}{3}{-7}\times\cvv{5}{1}{1}\right|\\
&=\frac12\left|\cvv{10}{-36}{-14}\right|=\sqrt{398}.
\end{align*}
\pt{(b)(i)} Since $\mathbf u+\mathbf v=\mathbf w$, and $|\mathbf u|=1$, $|\mathbf v|=2$ and $|\mathbf w|=3$, $\mathbf u$ and $\mathbf v$ must both be parallel to $\mathbf w$, as seen in the diagram.
\begin{center}\begin{tikzpicture}[>=Stealth,font=\small]
\draw[->,thick](0,0)--(1.5,0)node[midway,above]{$|\mathbf u|=1$};
\draw[->,thick](1.5,0)--(4.5,0)node[midway,above]{$|\mathbf v|=2$};
\draw[->,thick](0,-.5)--(4.5,-.5)node[midway,below]{$|\mathbf w|=3$};
\end{tikzpicture}\end{center}
Since $|\mathbf u|=1$, $\mathbf u$ is a unit vector parallel to $\mathbf w$.\par
\pt{(b)(ii)} Since each of the three vectors is perpendicular to the sum of the other two vectors,
\begin{align*}
\mathbf u\cdot(\mathbf v+\mathbf w)&=0,\qquad\mathbf v\cdot(\mathbf u+\mathbf w)=0,\qquad\mathbf w\cdot(\mathbf v+\mathbf u)=0\\
|\mathbf u+\mathbf v+\mathbf w|^2
&=(\mathbf u+\mathbf v+\mathbf w)\cdot(\mathbf u+\mathbf v+\mathbf w)\\
&=\mathbf u\cdot\mathbf u+\mathbf v\cdot\mathbf v+\mathbf w\cdot\mathbf w+\mathbf u\cdot(\mathbf v+\mathbf w)+\mathbf v\cdot(\mathbf u+\mathbf w)+\mathbf w\cdot(\mathbf v+\mathbf u)\\
&=|\mathbf u|^2+|\mathbf v|^2+|\mathbf w|^2\\
&=1^2+2^2+3^2=14.
\end{align*}
Since $|\mathbf u+\mathbf v+\mathbf w|>0$, $|\mathbf u+\mathbf v+\mathbf w|=\sqrt{14}$.}

\defQ{5}{RI 2025 Prelim P1 Q3}{%
Relative to the origin $O$, the points $A$ and $C$ have position vectors $\mathbf a$ and $2\mathbf b$ such that
\[\mathbf a=5p\mathbf i-2p\mathbf j+4p\mathbf k\quad\text{and}\quad\mathbf b=\mathbf i-2\mathbf j+2\mathbf k,\]
where $p$ is a positive constant. It is given that $|\mathbf a|=2|\mathbf b|$.
\begin{qparts}
\pt{(a)} Find the exact value of $p$.\mk{2}
\pt{(b)} Evaluate $(\mathbf a+2\mathbf b)\cdot(\mathbf a-2\mathbf b)$.\mk{1}
\pt{(c)} Evaluate $\mathbf a\times\mathbf b$ and hence find the area of triangle $OAC$.\mk{3}
\pt{(d)} Use a geometrical reason to explain why $|\mathbf a\times\mathbf b|=\dfrac14|\mathbf a+2\mathbf b||\mathbf a-2\mathbf b|$.\mk{2}
\end{qparts}}

\defS{5}{%
\pt{(a)} Since $|\mathbf a|=2|\mathbf b|$,
\begin{align*}
\sqrt{(5p)^2+(-2p)^2+(4p)^2}&=2\sqrt{1^2+(-2)^2+2^2}\\
3\sqrt5p&=6\qquad(p>0)\\
p&=\frac2{\sqrt5}=\frac{2\sqrt5}{5}.
\end{align*}
\pt{(b)}
\begin{align*}
(\mathbf a+2\mathbf b)\cdot(\mathbf a-2\mathbf b)
&=\mathbf a\cdot\mathbf a-2\mathbf a\cdot\mathbf b+2\mathbf b\cdot\mathbf a-4\mathbf b\cdot\mathbf b\\
&=|\mathbf a|^2-4|\mathbf b|^2\qquad(|\mathbf a|=2|\mathbf b|)\\
&=0.
\end{align*}
\pt{(c)}
\begin{align*}
\mathbf a\times\mathbf b&=\cvv{5p}{-2p}{4p}\times\cvv{1}{-2}{2}\\
&=p\cvv{5}{-2}{4}\times\cvv{1}{-2}{2}\\
&=p\cvv{4}{-6}{-8}=\frac{4\sqrt5}{5}\cvv{2}{-3}{-4}\\
\text{Area of triangle }OAC&=\frac12|\mathbf a\times2\mathbf b|=|\mathbf a\times\mathbf b|\\
&=\frac{4\sqrt5}{5}\sqrt{2^2+(-3)^2+(-4)^2}=\frac{4\sqrt{145}}5.
\end{align*}
\pt{(d)} Observe that $\vv{OD}=\mathbf a+2\mathbf b$ and $\vv{CA}=\mathbf a-2\mathbf b$. From (b), $OD$ is perpendicular to $CA$. In addition, $CA$ and $OD$, which are the diagonals of the parallelogram $OADC$, bisect each other.
\begin{center}\begin{tikzpicture}[scale=.8,font=\small,>=Stealth]
\coordinate(O)at(0,0);\coordinate(A)at(4,0);\coordinate(C)at(2,3.464);\coordinate(D)at(6,3.464);
\draw[semithick](O)--(A)--(D)--(C)--cycle;
\node[below left]at(O){$O$};\node[below right]at(A){$A$};\node[above left]at(C){$C$};\node[above right]at(D){$D$};
\draw[->](O)--(2,0)node[midway,below]{$\mathbf a$};\draw[->](O)--(1,1.732)node[midway,left]{$2\mathbf b$};
\end{tikzpicture}\end{center}
From (c), area of triangle $OAC=|\mathbf a\times\mathbf b|$. But area of triangle $OAC$ is also
\begin{align*}
\frac12(CA)\left(\frac12OD\right)&=\frac14|\mathbf a-2\mathbf b||\mathbf a+2\mathbf b|.
\end{align*}
Hence $|\mathbf a\times\mathbf b|=\dfrac14|\mathbf a+2\mathbf b||\mathbf a-2\mathbf b|$.}
