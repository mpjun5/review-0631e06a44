\Q{\textcolor{black!60}{\small[ASRJC 2025 Prelim P2 Q1]}\par\smallskip\textit{Official solution.}}
Since $AM:AB=3:4$, we have $AM:MB=3:1$. By the ratio theorem,
\[\vv{OM}=\frac{\mathbf a+3\mathbf b}{4}.\]
\begin{align*}
\text{Area of }\Delta OAM
&=\frac12\left|\vv{OA}\times\vv{OM}\right|
=\frac12\left|\mathbf a\times\frac{\mathbf a+3\mathbf b}{4}\right|
=\frac18\left|\mathbf a\times\mathbf a+3\mathbf a\times\mathbf b\right|
=\frac38|\mathbf a\times\mathbf b|\\
&=\frac38|\mathbf a||\mathbf b|\sin\frac{\pi}{6}
=\frac38\cdot k\cdot 1\cdot\frac12
=\frac{3k}{16}\text{ units}^2.
\end{align*}

\Q{\textcolor{black!60}{\small[NYJC 2025 Prelim P1 Q3]}\par\smallskip\textit{Official solution.}}
\par\smallskip\noindent\textbf{(a)}\par
$AB^2+BC^2=AC^2\Rightarrow|\vv{AB}|^2+|\vv{BC}|^2=|\vv{AC}|^2$.
Since $\vv{AB}+\vv{BC}=\vv{AC}$,
\begin{align*}
(\vv{AB}+\vv{BC})\cdot(\vv{AB}+\vv{BC})&=\vv{AC}\cdot\vv{AC}\\
|\vv{AB}|^2+2\vv{AB}\cdot\vv{BC}+|\vv{BC}|^2&=|\vv{AC}|^2\\
|\vv{AC}|^2+2\vv{AB}\cdot\vv{BC}&=|\vv{AC}|^2\quad\text{(using }AB^2+BC^2=AC^2\text{)}\\
\vv{AB}\cdot\vv{BC}&=0.
\end{align*}
So $\vv{AB}$ is perpendicular to $\vv{BC}$ and hence $\angle ABC=90^\circ$.
\par\smallskip\noindent\textbf{(b)}\par
By the ratio theorem, $\mathbf d=(1-\lambda)\mathbf a+\lambda\mathbf c$.
\begin{align*}
\text{Area of }\Delta ABD
&=\frac12\left|\vv{AB}\times\vv{BD}\right|
=\frac12\left|(\mathbf b-\mathbf a)\times(\mathbf d-\mathbf b)\right|\\
&=\frac12\left|(\mathbf b-\mathbf a)\times\big((1-\lambda)\mathbf a+\lambda\mathbf c-\mathbf b\big)\right|\\
&=\frac{\lambda}{2}\left|\mathbf a\times\mathbf b+\mathbf b\times\mathbf c+\mathbf c\times\mathbf a\right|.
\end{align*}
Hence $k=\dfrac{\lambda}{2}$.

\Q{\textcolor{black!60}{\small[NJC 2025 Prelim P2 Q3]}\par\smallskip\textit{Official solution.}}
\par\smallskip\noindent\textbf{(i)}\par
\[\mathbf p=\cvv{\cos t}{\sin t}{-1},\qquad\mathbf q=\cvv{\cos 2t}{-\sin 2t}{\alpha}.\]
For $O$, $P$, $Q$ collinear, $\mathbf p\times\mathbf q=\mathbf 0$. Equating the $\mathbf k$-component to $0$:
\begin{align*}
-\sin 2t\cos t-\cos 2t\sin t&=0\\
-\sin 3t&=0\qquad\Rightarrow\qquad t=\frac{\pi}{3}\ \text{or}\ \frac{2\pi}{3}\quad(0<t<\pi).
\end{align*}
When $t=\dfrac{\pi}{3}$: $-\big(\alpha\cos\frac{\pi}{3}+\cos\frac{2\pi}{3}\big)=0\Rightarrow\alpha=1$.\\
When $t=\dfrac{2\pi}{3}$: $-\big(\alpha\cos\frac{2\pi}{3}+\cos\frac{4\pi}{3}\big)=0\Rightarrow\alpha=-1$.\\
Final answers: $t=\dfrac{\pi}{3}$, $\alpha=1$; \ $t=\dfrac{2\pi}{3}$, $\alpha=-1$.
\par\smallskip\noindent\textbf{(ii)}\par
\begin{align*}
\mathbf p\cdot\mathbf q&=\cos t\cos 2t-\sin t\sin 2t-0.5=\cos 3t-0.5\quad\text{(shown)}.\\
\text{Length of projection of }\mathbf q\text{ onto }\mathbf p
&=\frac{|\mathbf p\cdot\mathbf q|}{|\mathbf p|}
=\frac{|\cos 3t-0.5|}{\sqrt{2}}.
\end{align*}
Greatest length when $\cos 3t=-1$:
\[\frac{|-1-0.5|}{\sqrt{2}}=\frac{1.5}{\sqrt{2}}=\frac{3}{2\sqrt{2}}\quad(=1.06\text{ to 3 s.f.}).\]

\Q{\textcolor{black!60}{\small[CJC 2025 Promo Q6]}\par\smallskip\textit{Official solution.}}
\par\smallskip\noindent\textbf{(a)}\par
Since $AB$ is a diameter through $O$, $\mathbf b=-\mathbf a$.
\begin{align*}
\vv{AC}&=\mathbf c-\mathbf a,\qquad\vv{BC}=\mathbf c-\mathbf b=\mathbf c+\mathbf a\\
\vv{AC}\cdot\vv{BC}&=(\mathbf c-\mathbf a)\cdot(\mathbf c+\mathbf a)=|\mathbf c|^2-|\mathbf a|^2=0
\end{align*}
(since $|\mathbf c|=|\mathbf a|=$ radius). Hence $AC$ and $BC$ are perpendicular.
\par\smallskip\noindent\textbf{(b)}\par
$|\mathbf a\cdot(\mathbf c-\mathbf a)|$ is the length of the projection of $\vv{AC}$ onto $\vv{OA}$.
\par\smallskip\noindent\textbf{(c)}\par
By the ratio theorem, $\vv{OM}=\dfrac{\vv{OD}+\vv{OA}}{2}=\dfrac12(\mathbf c+2\mathbf a+\mathbf a)=\dfrac12\mathbf c+\dfrac32\mathbf a$
(since $\vv{OD}=\mathbf c+2\mathbf a$ with $\mathbf b=-\mathbf a$). Also $\vv{OR}=3\mathbf a$.
\begin{align*}
\vv{CM}\times\vv{CR}
&=\left(\vv{OM}-\mathbf c\right)\times\left(\vv{OR}-\mathbf c\right)
=\left(\tfrac12\mathbf c+\tfrac32\mathbf a-\mathbf c\right)\times(3\mathbf a-\mathbf c)\\
&=\left(\tfrac32\mathbf a-\tfrac12\mathbf c\right)\times(3\mathbf a-\mathbf c)
=\tfrac92(\mathbf a\times\mathbf a)-\tfrac32(\mathbf a\times\mathbf c)-\tfrac32(\mathbf c\times\mathbf a)+\tfrac12(\mathbf c\times\mathbf c)\\
&=0-\tfrac32\mathbf a\times\mathbf c+\tfrac32\mathbf a\times\mathbf c+0=\mathbf 0.
\end{align*}
Hence $\vv{CM}$ and $\vv{CR}$ are parallel. Since $C$ is a common point, $C$, $M$ and $R$ are collinear.

Alternatively: $\vv{CM}=\tfrac32\mathbf a-\tfrac12\mathbf c$ and $\vv{CR}=3\mathbf a-\mathbf c=2\vv{CM}$.
