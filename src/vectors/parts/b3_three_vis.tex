\subsubsection*{W3.6 \ Summary}
\begin{center}
\renewcommand{\arraystretch}{1.5}
\begin{tabular}{|l|l|l|}
\hline
\textbf{Case} & \textbf{Condition} & \textbf{GC (PlySmlt2)} \\ \hline
1. all parallel & $\mathbf{n}_1\parallel\mathbf{n}_2\parallel\mathbf{n}_3$ & no solution \\ \hline
2. two parallel, one not & $\mathbf{n}_1\parallel\mathbf{n}_2$, $\mathbf{n}_3\nparallel\mathbf{n}_1$ & no solution \\ \hline
3. one point & $\mathbf{b}\cdot\mathbf{n}_3\neq0$ & unique solution \\ \hline
4. common line & $\mathbf{b}\cdot\mathbf{n}_3=0$, $\mathbf{a}\cdot\mathbf{n}_3=d_3$ & infinitely many, 1 parameter \\ \hline
5. triangular prism & $\mathbf{b}\cdot\mathbf{n}_3=0$, $\mathbf{a}\cdot\mathbf{n}_3\neq d_3$ & no solution \\ \hline
\end{tabular}
\end{center}

\begin{example}
$p_1: x+y+z=6$, $p_2: 2x-y+z=3$, $l=p_1\cap p_2: \mathbf{r}=\col{3}{3}{0}+t\col{2}{1}{-3}$. Describe the intersection of $p_1$, $p_2$, $p_3$ for (a) $p_3: x+2y=5$, (b) $p_3: x-2y=-3$.
\end{example}
\begin{vis}
\begin{tikzpicture}[v3=0.55]
\vaxes{7}{7}{7}
\fill[pp] (6,0,0)--(0,6,0)--(0,0,6)--cycle;
\fill[pr] (5,0,-1)--(-1,3,-1)--(-1,3,6)--(5,0,6)--cycle;
\draw[ln,vC] (5,4,-3)--(-1,1,6) node[above]{$l$};
\node[dot,label=right:{$(1,2,3)$}] at (1,2,3){};
\end{tikzpicture}
\end{vis}
\solution{
\begin{align*}
\text{(a)}\quad \col{3+2t}{3+t}{-3t}\cdot\col{1}{2}{0} &= 5\\
9+4t &= 5\\
t &= -1\\
\therefore \text{one point } &(1,2,3)\\
\text{(b)}\quad \col{2}{1}{-3}\cdot\col{1}{-2}{0} &= 0\\
\col{3}{3}{0}\cdot\col{1}{-2}{0} &= -3\\
\therefore l &\subset p_3\\
\therefore \text{common line } &l
\end{align*}
}
