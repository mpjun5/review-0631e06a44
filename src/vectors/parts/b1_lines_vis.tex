\end{vis}
\begin{align*}
\mathbf{b}_1 = k\mathbf{b}_2, \ A_1\notin l_2 &\Rightarrow \text{parallel}\\
\mathbf{b}_1 = k\mathbf{b}_2, \ A_1\in l_2 &\Rightarrow \text{same line}\\
\mathbf{b}_1 \neq k\mathbf{b}_2, \ \text{unique } (\lambda,\mu) &\Rightarrow \text{intersecting}\\
\mathbf{b}_1 \neq k\mathbf{b}_2, \ \text{no } (\lambda,\mu) &\Rightarrow \text{skew}
\end{align*}

\begin{example}
$l_1:\mathbf{r}=\col{1}{0}{2}+\lambda\col{1}{2}{1}$, $l_2:\mathbf{r}=\col{1}{2}{4}+\mu\col{1}{0}{-1}$. Show $l_1$, $l_2$ intersect; find the point and the angle.
\end{example}
\begin{vis}
\begin{tikzpicture}[v3=1.1]
\vaxes{3.5}{3.5}{4.5}
\draw[ln,vA] (0.5,-1,1.5)--(3,4,4) node[right]{$l_1$};
\draw[ln,vB] (0.5,2,4.5)--(3.5,2,1.5) node[right]{$l_2$};
\node[dot,label=above:{$(2,2,3)$}] at (2,2,3){};
\end{tikzpicture}
\end{vis}
\solution{
\begin{align*}
\col{1}{2}{1} &\neq k\col{1}{0}{-1} \ \therefore \text{not parallel}\\
1+\lambda &= 1+\mu \quad (1)\\
2\lambda &= 2 \quad (2)\\
2+\lambda &= 4-\mu \quad (3)\\
\text{From (1), (2): } \lambda &= 1, \ \mu = 1\\
\text{(3): LHS} &= 3, \ \text{RHS} = 4-1 = 3\\
\therefore \text{LHS} &= \text{RHS}\\
\therefore \text{intersect at } &(2,2,3)\\
\cos\theta &= \frac{\left|\col{1}{2}{1}\cdot\col{1}{0}{-1}\right|}{\sqrt6\sqrt2}\\
&= 0\\
\theta &= 90^\circ
\end{align*}
}

\begin{align*}
\overrightarrow{ON} &= \col{1+3\lambda}{2}{3-\lambda}\\
\overrightarrow{AN} &= \col{-2+3\lambda}{4}{4-\lambda}\\
\overrightarrow{AN}\cdot\col{3}{0}{-1} &= 0\\
10\lambda-10 &= 0\\
\lambda &= 1\\
\therefore N &= (4,2,2)\\
\overrightarrow{ON} &= \frac{\overrightarrow{OA}+\overrightarrow{OA'}}{2}\\
\overrightarrow{OA'} &= 2\overrightarrow{ON}-\overrightarrow{OA}\\
&= 2\col{4}{2}{2}-\col{3}{-2}{-1}\\
&= \col{5}{6}{5}\\
AN &= \left|\col{1}{4}{3}\right|\\
&= \sqrt{26}
\end{align*}
