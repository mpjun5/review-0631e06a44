\subsubsection*{V2.1 \ Angle $\theta$: tails together}
\begin{vis}
\begin{tikzpicture}[v3=1.3]
\vaxes{3.2}{3}{3}
\coordinate (O) at (0,0,0); \coordinate (A) at (0.3,2,2); \coordinate (B) at (2.6,0,0.4);
\draw[va] (0,0,0)--(A) node[above]{$\mathbf{a}$};
\draw[vb] (0,0,0)--(B) node[right]{$\mathbf{b}$};
\vang{O}{B}{A}{$\theta$}
\end{tikzpicture}
\end{vis}
\begin{align*}
\mathbf{a}\cdot\mathbf{b} &= a_1b_1+a_2b_2+a_3b_3\\
\mathbf{a}\cdot\mathbf{b} &= |\mathbf{a}||\mathbf{b}|\cos\theta\\
\cos\theta &= \frac{\mathbf{a}\cdot\mathbf{b}}{|\mathbf{a}||\mathbf{b}|}
\end{align*}

\text{projection vector} &= (\mathbf{a}\cdot\hat{\mathbf{b}})\hat{\mathbf{b}}
\end{align*}

\subsubsection*{V2.4 \ Projection of $\mathbf{a}$ onto $\mathbf{b}$ $\neq$ projection of $\mathbf{b}$ onto $\mathbf{a}$}
\begin{vis}
\begin{tikzpicture}[v3=1.2]
\coordinate (O) at (0,0,0); \coordinate (A) at (1,1.5,2); \coordinate (B) at (3,0,0);
\draw[vb] (O)--(B) node[right]{$\mathbf{b}$}; \draw[va] (O)--(A) node[above]{$\mathbf{a}$};
\draw[dsh] (A)--(1,0,0); \draw[vc, line width=2.2pt] (O)--(1,0,0);
\end{tikzpicture}\qquad
\begin{tikzpicture}[v3=1.2]
\coordinate (O) at (0,0,0); \coordinate (A) at (1,1.5,2); \coordinate (B) at (3,0,0);
\coordinate (G) at ($(O)!0.4138!(A)$);
\draw[vb] (O)--(B) node[right]{$\mathbf{b}$}; \draw[va] (O)--(A) node[above]{$\mathbf{a}$};
\draw[dsh] (B)--(G); \draw[vc, line width=2.2pt] (O)--(G);
\end{tikzpicture}
\end{vis}
\[
|\mathbf{a}\cdot\hat{\mathbf{b}}| = \frac{|\mathbf{a}\cdot\mathbf{b}|}{|\mathbf{b}|}, \qquad |\mathbf{b}\cdot\hat{\mathbf{a}}| = \frac{|\mathbf{a}\cdot\mathbf{b}|}{|\mathbf{a}|}
\]

\begin{example}
$\mathbf{a}=\col{1}{2}{2}$, $\mathbf{b}=\col{2}{-1}{2}$. Find $\angle$ between $\mathbf{a}$ and $\mathbf{b}$, the length of projection of $\mathbf{a}$ onto $\mathbf{b}$, and the projection vector.
\end{example}
\begin{vis}
\begin{tikzpicture}[v3=1.4]
\vaxes{3}{2.6}{2.8}
\coordinate (O) at (0,0,0); \coordinate (A) at (1,2,2); \coordinate (B) at (2,-1,2); \coordinate (F) at (8/9,-4/9,8/9);
\draw[vb] (O)--(B) node[right]{$\mathbf{b}$}; \draw[va] (O)--(A) node[above]{$\mathbf{a}$};
\draw[dsh] (A)--(F); \draw[vc, line width=2.2pt] (O)--(F);
\vang{O}{B}{A}{$\theta$}
\end{tikzpicture}
\end{vis}
\solution{
\begin{align*}
\mathbf{a}\cdot\mathbf{b} &= 2-2+4\\
&= 4\\
|\mathbf{a}| &= 3, \quad |\mathbf{b}| = 3\\
\cos\theta &= \frac{4}{3(3)}\\
\theta &= \cos^{-1}\frac{4}{9}\\
&= 63.6^\circ \ (1\text{ d.p.})\\
|\mathbf{a}\cdot\hat{\mathbf{b}}| &= \frac{4}{3}\\
(\mathbf{a}\cdot\hat{\mathbf{b}})\hat{\mathbf{b}} &= \frac{4}{3}\cdot\frac{1}{3}\col{2}{-1}{2}\\
&= \frac{4}{9}\col{2}{-1}{2}
\end{align*}
}

\subsubsection*{V2.5 \ Angle with an axis}
\begin{vis}
\begin{tikzpicture}[v3=1.4]
\vaxes{2.6}{2.6}{2.6}
\coordinate (O) at (0,0,0); \coordinate (A) at (1,2,2); \coordinate (I) at (1,0,0);
\draw[va] (O)--(A) node[above]{$\mathbf{a}$};
\draw[vb] (O)--(I) node[below]{$\mathbf{i}$};
\vang{O}{I}{A}{$\alpha$}
\end{tikzpicture}
\end{vis}
\begin{align*}
\cos\alpha &= \frac{\mathbf{a}\cdot\mathbf{i}}{|\mathbf{a}||\mathbf{i}|}\\
&= \frac{1}{3}\\
\alpha &= 70.5^\circ \ (1\text{ d.p.})
\end{align*}
