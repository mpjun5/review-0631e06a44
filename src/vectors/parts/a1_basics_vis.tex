\subsubsection*{V1.6 \ Displacement $\overrightarrow{AB}=\mathbf{b}-\mathbf{a}$}

\overrightarrow{AB} &= \overrightarrow{OB}-\overrightarrow{OA}\\
&= \col{4}{2}{3}-\col{1}{2}{0}\\
&= \col{3}{0}{3}\\
AB &= \sqrt{3^2+0^2+3^2}\\
&= 3\sqrt{2}
\end{align*}

\subsubsection*{V1.10 \ Ratio theorem}
\begin{vis}
\begin{tikzpicture}[v3=1.15]
\vaxes{5}{4}{4}
\coordinate (A) at (1,2,0); \coordinate (B) at (4,0.5,3);
\coordinate (C) at ($(A)!0.4!(B)$);
\draw[ln, black!60] (A)--(B);
\draw[va] (0,0,0)--(A) node[pos=0.5,left]{$\mathbf{a}$};
\draw[vb] (0,0,0)--(B) node[pos=0.5,below right]{$\mathbf{b}$};
\draw[vc] (0,0,0)--(C) node[pos=0.6,above]{$\mathbf{c}$};
\node[dot,label=left:$A$] at (A){};\node[dot,label=right:$B$] at (B){};\node[dot,label=above:$C$] at (C){};
\path (A)--(C) node[midway,above]{$\mu$}; \path (C)--(B) node[midway,above]{$\lambda$};
\end{tikzpicture}
\quad
\begin{tikzpicture}[v3=1.15]
\coordinate (D) at (0,3,3.5); \coordinate (A) at (1,2,0); \coordinate (B) at (4,0.5,3);
\coordinate (C) at ($(A)!0.4!(B)$);
\draw[ln, black!60] (A)--(B);
\draw[va] (D)--(A); \draw[vb] (D)--(B); \draw[vc] (D)--(C);
\node[dot,label=above:$D$] at (D){};\node[dot,label=left:$A$] at (A){};\node[dot,label=right:$B$] at (B){};\node[dot,label=below:$C$] at (C){};
\path (A)--(C) node[midway,below]{$\mu$}; \path (C)--(B) node[midway,below]{$\lambda$};
\end{tikzpicture}
\end{vis}
\begin{align*}
AC:CB &= \mu:\lambda\\
\mathbf{c} &= \frac{\lambda\mathbf{a}+\mu\mathbf{b}}{\lambda+\mu}\\
\overrightarrow{DC} &= \frac{\lambda\overrightarrow{DA}+\mu\overrightarrow{DB}}{\lambda+\mu}\\
\text{midpoint: } \mathbf{m} &= \frac{\mathbf{a}+\mathbf{b}}{2}
\end{align*}

\begin{example}
$A(1,2,3)$, $B(7,-1,6)$. $C$ on $AB$, $AC:CB=1:2$. Find $\overrightarrow{OC}$.
\end{example}
\begin{vis}
\begin{tikzpicture}[v3=0.75]
\vaxes{8}{4}{7}
\coordinate (A) at (1,2,3); \coordinate (B) at (7,-1,6); \coordinate (C) at (3,1,4);
\draw[ln, black!60] (A)--(B);
\draw[va] (0,0,0)--(A); \draw[vb] (0,0,0)--(B); \draw[vc] (0,0,0)--(C);
\node[dot,label=above:{$A(1,2,3)$}] at (A){};\node[dot,label=right:{$B(7,-1,6)$}] at (B){};\node[dot,label=below right:{$C$}] at (C){};
\path (A)--(C) node[midway,above]{$1$}; \path (C)--(B) node[midway,above]{$2$};
\end{tikzpicture}
\end{vis}
\solution{
\begin{align*}
\overrightarrow{OC} &= \frac{2\overrightarrow{OA}+\overrightarrow{OB}}{1+2}\\
&= \frac{1}{3}\left[2\col{1}{2}{3}+\col{7}{-1}{6}\right]\\
&= \frac{1}{3}\col{9}{3}{12}\\
&= \col{3}{1}{4}
\end{align*}
}
