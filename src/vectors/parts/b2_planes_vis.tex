\begin{example}
$l:\mathbf{r}=\col{1}{0}{2}+\lambda\col{1}{2}{1}$, $p: \mathbf{r}\cdot\col{1}{1}{1}=10$. Find the point of intersection and the acute angle between $l$ and $p$.
\end{example}
\begin{vis}
\begin{tikzpicture}[v3=0.8]
\vaxes{5}{5}{5}
\fill[pp] (10,0,0)--(0,10,0)--(0,0,10)--cycle;
\draw[ln,vB] (1,0,2)--(11/4,7/2,15/4);
\draw[ln,vB,dashed] (11/4,7/2,15/4)--(3.5,5,4.5) node[right]{$l$};
\node[dot,label=left:{$\left(\frac{11}{4},\frac72,\frac{15}{4}\right)$}] at (11/4,7/2,15/4){};
\node[dot,label=below:{$(1,0,2)$}] at (1,0,2){};
\end{tikzpicture}
\end{vis}
\solution{
\begin{align*}
\col{1+\lambda}{2\lambda}{2+\lambda}\cdot\col{1}{1}{1} &= 10\\
3+4\lambda &= 10\\
\lambda &= \frac74\\
\therefore \text{point } &\left(\frac{11}{4},\frac72,\frac{15}{4}\right)\\
\sin\phi &= \frac{\left|\col{1}{2}{1}\cdot\col{1}{1}{1}\right|}{\sqrt6\sqrt3}\\
&= \frac{4}{\sqrt{18}}\\
\phi &= 70.5^\circ \ (1\text{ d.p.})
\end{align*}
}

\begin{vis}

\begin{tikzpicture}[v3=0.85]
\fill[pp] (-1,-1,0)--(3,-1,0)--(3,3,0)--(-1,3,0)--cycle;
\fill[pq] (-1,-1,1.2)--(3,-1,1.2)--(3,3,1.2)--(-1,3,1.2)--cycle;
\draw[vn] (2.5,-0.5,0)--(2.5,-0.5,0.9);
\node[below=2pt] at (current bounding box.south){$\mathbf{n}_1\parallel\mathbf{n}_2$, parallel};
\end{tikzpicture}\quad
\begin{tikzpicture}[v3=0.85]
\fill[pp] (-1,-1,0)--(3,-1,0)--(3,3,0)--(-1,3,0)--cycle;
\fill[pq, fill opacity=0.4] (-1,-1,0)--(3,-1,0)--(3,3,0)--(-1,3,0)--cycle;
\node[below=2pt] at (current bounding box.south){same plane};
\end{tikzpicture}
\end{vis}
\begin{align*}
\mathbf{n}_1 \neq k\mathbf{n}_2 &\Rightarrow \text{line, direction } \mathbf{n}_1\times\mathbf{n}_2\\
\mathbf{n}_1 = k\mathbf{n}_2, \ d_1 \neq kd_2 &\Rightarrow \text{parallel}\\
\mathbf{n}_1 = k\mathbf{n}_2, \ d_1 = kd_2 &\Rightarrow \text{same plane}
\end{align*}

\begin{example}
$p_1: x+y+z=6$, $p_2: x-y+2z=2$. Find the line of intersection and the acute angle between $p_1$ and $p_2$.
\end{example}
\begin{vis}
\begin{tikzpicture}[v3=0.6]
\vaxes{7}{7}{7}
\fill[pp] (6,0,0)--(0,6,0)--(0,0,6)--cycle;
\fill[pq] (2,0,0)--(8,0,-3)--(6,4,0)--(0,4,3)--cycle;
\draw[ln,vC] (1,3,2)--(7,1,-2) node[right]{$l$};
\node[dot,label=above right:{$(4,2,0)$}] at (4,2,0){};
\end{tikzpicture}
\end{vis}
\solution{
\begin{align*}
\mathbf{n}_1\times\mathbf{n}_2 &= \col{1}{1}{1}\times\col{1}{-1}{2}\\
&= \col{1(2)-1(-1)}{-(1(2)-1(1))}{1(-1)-1(1)}\\
&= \col{3}{-1}{-2}\\
\text{Let } z &= 0\\
x+y &= 6\\
x-y &= 2\\
\therefore x &= 4, \ y = 2\\
l: \mathbf{r} &= \col{4}{2}{0}+\lambda\col{3}{-1}{-2}, \ \lambda\in\RR\\
\cos\theta &= \frac{\left|\col{1}{1}{1}\cdot\col{1}{-1}{2}\right|}{\sqrt3\sqrt6}\\
&= \frac{2}{\sqrt{18}}\\
\theta &= 61.9^\circ \ (1\text{ d.p.})
\end{align*}
}

\subsubsection*{W2.11 \ Distance between parallel planes}
\begin{vis}
\begin{tikzpicture}[v3=1.1]
\fill[pp] (-1,-1,0)--(3,-1,0)--(3,3,0)--(-1,3,0)--cycle;
\fill[pq] (-1,-1,1.8)--(3,-1,1.8)--(3,3,1.8)--(-1,3,1.8)--cycle;
\coordinate (A) at (0,0,0); \coordinate (B) at (1.8,1.6,1.8);
\draw[vb] (A)--(B) node[pos=0.5,left]{$\overrightarrow{AB}$};
\draw[vB, very thick] (B)--(1.8,1.6,0);
\node[dot,label=left:$A$] at (A){}; \node[dot,label=above:$B$] at (B){};
\node at (3.2,3.2,0.1){$p_1$}; \node at (3.2,3.2,1.9){$p_2$};
\end{tikzpicture}
\end{vis}
\begin{align*}
\text{distance} &= |\overrightarrow{AB}\cdot\hat{\mathbf{n}}|\\
&= \frac{|d_2-d_1|}{|\mathbf{n}|} \quad (\text{same } \mathbf{n})\\
p_1: \mathbf{r}\cdot\col{1}{1}{2}=1, \ p_2: \mathbf{r}\cdot\col{1}{1}{2}=13 &\Rightarrow \frac{|13-1|}{\sqrt6} = 2\sqrt6
\end{align*}
